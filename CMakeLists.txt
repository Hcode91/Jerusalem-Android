cmake_minimum_required(VERSION 3.22)
project(jerusalem_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JERUSALEM_SANITIZE "Enable host ASan and UBSan" OFF)
add_library(jerusalem_core STATIC src/resources.cpp)
set_target_properties(jerusalem_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(jerusalem_core PUBLIC include)
target_compile_options(jerusalem_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(JERUSALEM_SANITIZE AND NOT ANDROID)
  target_compile_options(jerusalem_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jerusalem_core PUBLIC -fsanitize=address,undefined)
endif()
if(ANDROID)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL/CMakeLists.txt")
    message(FATAL_ERROR "SDL source missing. Run the GitHub Actions workflow to provision it.")
  endif()
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)
  add_library(main SHARED src/android_main.cpp)
  target_link_libraries(main PRIVATE jerusalem_core SDL3::SDL3 log)
  target_compile_options(main PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
  add_executable(jinspect tools/jinspect.cpp)
  target_link_libraries(jinspect PRIVATE jerusalem_core)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(resource_tests tests/resource_tests.cpp)
    target_link_libraries(resource_tests PRIVATE jerusalem_core)
    add_test(NAME resources COMMAND resource_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME pe_probe COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
    set_tests_properties(pe_probe PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
