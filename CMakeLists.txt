cmake_minimum_required(VERSION 3.22)
project(jerusalem_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JERUSALEM_SANITIZE "Enable host ASan and UBSan" OFF)
option(JERUSALEM_DESKTOP_VIEWER "Build the SDL2 desktop panorama viewer" OFF)
add_library(jerusalem_core STATIC src/resources.cpp src/audio_codec.cpp src/aqc_codec.cpp src/aqc_animation.cpp src/panorama.cpp src/navigation.cpp src/scene_audio.cpp src/interactions.cpp src/checkpoint.cpp src/dialogue.cpp src/cinema.cpp src/stage.cpp src/a01_story.cpp src/story_save.cpp src/menu.cpp src/story_assets.cpp src/x86.cpp src/script_area.cpp src/script_story.cpp)
set_target_properties(jerusalem_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(jerusalem_core PUBLIC include)
target_compile_options(jerusalem_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(JERUSALEM_SANITIZE AND NOT ANDROID)
  target_compile_options(jerusalem_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jerusalem_core PUBLIC -fsanitize=address,undefined)
endif()
if(ANDROID)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL/CMakeLists.txt")
    message(FATAL_ERROR "SDL source missing. Run the GitHub Actions workflow to provision it.")
  endif()
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)
  add_library(main SHARED src/android_main.cpp)
  target_link_libraries(main PRIVATE jerusalem_core SDL3::SDL3 log)
  target_compile_options(main PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
  add_executable(jinspect tools/jinspect.cpp)
  target_link_libraries(jinspect PRIVATE jerusalem_core)
  add_executable(script_probe tools/script_probe.cpp)
  target_link_libraries(script_probe PRIVATE jerusalem_core)
  if(JERUSALEM_DESKTOP_VIEWER)
    find_package(SDL2 REQUIRED CONFIG)
    add_executable(jerusalem_viewer tools/panorama_viewer.cpp)
    target_link_libraries(jerusalem_viewer PRIVATE jerusalem_core SDL2::SDL2)
    add_executable(jerusalem_cinema tools/cinema_viewer.cpp)
    target_link_libraries(jerusalem_cinema PRIVATE jerusalem_core SDL2::SDL2)
    target_compile_options(jerusalem_cinema PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(jerusalem_viewer PRIVATE SDL2::SDL2main)
      target_link_libraries(jerusalem_cinema PRIVATE SDL2::SDL2main)
    endif()
    target_compile_options(jerusalem_viewer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  include(CTest)
  if(BUILD_TESTING)
    add_executable(resource_tests tests/resource_tests.cpp)
    target_link_libraries(resource_tests PRIVATE jerusalem_core)
    add_test(NAME resources COMMAND resource_tests)
    add_executable(panorama_tests tests/panorama_tests.cpp)
    target_link_libraries(panorama_tests PRIVATE jerusalem_core)
    add_test(NAME panorama COMMAND panorama_tests)
    add_executable(navigation_tests tests/navigation_tests.cpp)
    target_link_libraries(navigation_tests PRIVATE jerusalem_core)
    add_test(NAME navigation COMMAND navigation_tests)
    add_executable(interaction_tests tests/interaction_tests.cpp)
    target_link_libraries(interaction_tests PRIVATE jerusalem_core)
    add_test(NAME interactions COMMAND interaction_tests)
    add_executable(checkpoint_tests tests/checkpoint_tests.cpp)
    target_link_libraries(checkpoint_tests PRIVATE jerusalem_core)
    add_test(NAME checkpoint COMMAND checkpoint_tests)
    add_executable(story_tests tests/story_tests.cpp)
    target_link_libraries(story_tests PRIVATE jerusalem_core)
    add_test(NAME story_media COMMAND story_tests)
    add_executable(stage_tests tests/stage_tests.cpp)
    target_link_libraries(stage_tests PRIVATE jerusalem_core)
    add_test(NAME stage_story COMMAND stage_tests)
    set_tests_properties(stage_story PROPERTIES ENVIRONMENT "JERUSALEM_EXPORT_CAMP=${CMAKE_BINARY_DIR}/a01-smoke-model")
    add_executable(script_tests tests/script_tests.cpp)
    target_link_libraries(script_tests PRIVATE jerusalem_core)
    add_test(NAME script COMMAND script_tests)
    add_executable(animation_tests tests/animation_tests.cpp)
    target_link_libraries(animation_tests PRIVATE jerusalem_core)
    add_test(NAME animation COMMAND animation_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    if(TARGET jerusalem_viewer)
      add_test(NAME desktop_viewer COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/viewer_smoke.py" $<TARGET_FILE:jerusalem_viewer> $<TARGET_FILE:jerusalem_cinema>)
    endif()
    add_test(NAME pe_probe COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
    set_tests_properties(pe_probe PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(pe_probe PROPERTIES ENVIRONMENT "JERUSALEM_JINSPECT=$<TARGET_FILE:jinspect>;JERUSALEM_CAMP_MODEL=${CMAKE_BINARY_DIR}/a01-smoke-model")
  endif()
endif()
